    function isContentEditor() {
      return request.auth != null
        && request.auth.token.email in ['info-team@researchcops.com', 'deep.cruise@gmail.com', 'deepak.monga@researchcops.com'];
    }

    // siteContent/published is read by the website build without signing in.
    // siteContent/draft is the editors' work in progress.
    match /siteContent/{docId} {
      allow read: if docId == 'published' || isContentEditor();
      allow write: if isContentEditor();
    }

    // Every published version, for History → Load into editor
    match /siteContentVersions/{versionId} {
      allow read: if isContentEditor();
      allow create: if isContentEditor();
      allow update, delete: if false;
    }
